        // 参照を残さないよう、別メソッドで数えて値だけ返す
        [MethodImpl(MethodImplOptions.NoInlining)]
        static string Count(string tag)
        {
            int modules=0,enemies=0;long bytes=0;
            foreach(var t in Resources.FindObjectsOfTypeAll<Texture2D>())
            {
                bool module=ModuleNames.Contains(t.name);
                bool enemy=t.name.StartsWith("enemy-");
                if(module)modules++;if(enemy)enemies++;
                if(module||enemy)bytes+=Profiler.GetRuntimeMemorySizeLong(t);
            }
            return $"{tag}: modules={modules} enemies={enemies} {bytes/1048576}MB";
        }
        [MethodImpl(MethodImplOptions.NoInlining)]
